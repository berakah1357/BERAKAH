async function toggleFollow() { const button = document.getElementById("followButton"); const status = document.getElementById("status"); button.disabled = true; status.textContent = ""; try { const following = await checkFollowing(); /* ========================= UNFOLLOW ========================= */ if (following) { const response = await fetch( SUPABASE_URL + "/rest/v1/follows" + "?follower_id=eq." + encodeURIComponent(currentUser.id) + "&following_id=eq." + encodeURIComponent(profileUserId), { method: "DELETE", headers: headers() } ); const text = await response.text(); if (!response.ok) { throw new Error( text || "Could not unfollow." ); } status.textContent = "Unfollowed successfully."; } /* ========================= FOLLOW ========================= */ else { const response = await fetch( SUPABASE_URL + "/rest/v1/follows", { method: "POST", headers: headers({ "Content-Type": "application/json", "Prefer": "return=minimal" }), body: JSON.stringify({ follower_id: currentUser.id, following_id: profileUserId }) } ); const text = await response.text(); if (!response.ok) { throw new Error( text || "Could not follow." ); } /* ========================= CREATE NOTIFICATION ========================= */ const actorName = currentUser .user_metadata ?.username || currentUser .email ?.split("@")[0] || "BERAKAH User"; const notificationResponse = await fetch( SUPABASE_URL + "/rest/v1/notifications", { method: "POST", headers: headers({ "Content-Type": "application/json", "Prefer": "return=minimal" }), body: JSON.stringify({ user_id: profileUserId, actor_id: currentUser.id, type: "follow", message: actorName + " follows you", read: false }) } ); if (!notificationResponse.ok) { console.error( "Follow worked, but notification failed:", await notificationResponse.text() ); } status.textContent = "Following successfully."; } await loadFollowCounts(); await updateFollowButton(); } catch(error) { console.error(error); status.textContent = error.message; } button.disabled = false; }
